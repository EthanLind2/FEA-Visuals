function linspace(a, b, n) { return Array.from({length: n}, (_, k) => a + (b - a) * k / (n - 1)); }

// Legendre polynomial P_n(x) and its derivative
function legendre(n, x) {
  if (n === 0) return [1, 0];
  let p0 = 1, p1 = x;
  for (let k = 2; k <= n; k++) { const p2 = ((2*k - 1) * x * p1 - (k - 1) * p0) / k; p0 = p1; p1 = p2; }
  return [p1, n * (x * p1 - p0) / (x * x - 1)];
}

// Gauss–Legendre points and weights on [-1, 1]
function gauss(n) {
  const pts = [];
  for (let i = 0; i < n; i++) {
    let x = Math.cos(Math.PI * (i + 0.75) / (n + 0.5));
    for (let it = 0; it < 100; it++) {
      const [P, dP] = legendre(n, x);
      const dx = P / dP; x -= dx;
      if (Math.abs(dx) < 1e-15) break;
    }
    const [, dP] = legendre(n, x);
    pts.push({xi: x, w: 2 / ((1 - x * x) * dP * dP)});
  }
  return pts.sort((a, b) => a.xi - b.xi);
}

// Node sets on the reference element [-1, 1] (p+1 nodes, ordered left to right)
function refNodes(p, type) {
  if (type === "Chebyshev–Lobatto") return d3.range(p + 1).map(k => -Math.cos(k * Math.PI / p));
  if (type === "Gauss–Lobatto–Legendre") {
    const x = [-1];
    for (let k = 1; k < p; k++) {
      let z = -Math.cos(k * Math.PI / p);
      for (let it = 0; it < 100; it++) {
        const [P, dP] = legendre(p, z);
        const d2P = (2 * z * dP - p * (p + 1) * P) / (1 - z * z);
        const dz = dP / d2P; z -= dz;
        if (Math.abs(dz) < 1e-15) break;
      }
      x.push(z);
    }
    x.push(1);
    return x;
  }
  return d3.range(p + 1).map(k => -1 + 2 * k / p);   // equally spaced
}

// Lagrange basis N_i(xi) = prod_{k != i} (xi - xi_k)/(xi_i - xi_k)
function lagrange(nodes, i, x) {
  let v = 1;
  for (let k = 0; k < nodes.length; k++) if (k !== i) v *= (x - nodes[k]) / (nodes[i] - nodes[k]);
  return v;
}
// dN_i/dxi via the product rule
function dlagrange(nodes, i, x) {
  let s = 0;
  for (let m = 0; m < nodes.length; m++) {
    if (m === i) continue;
    let t = 1 / (nodes[i] - nodes[m]);
    for (let k = 0; k < nodes.length; k++) if (k !== i && k !== m) t *= (x - nodes[k]) / (nodes[i] - nodes[k]);
    s += t;
  }
  return s;
}

function fmt(v, d = 4) {
  if (Math.abs(v) < 1e-10) return "0";
  return (+v.toPrecision(d)).toString();
}

// Dense Gaussian elimination with partial pivoting
function solveDense(A, b) {
  const n = b.length, M = A.map((r, i) => [...r, b[i]]);
  for (let c = 0; c < n; c++) {
    let piv = c;
    for (let r = c + 1; r < n; r++) if (Math.abs(M[r][c]) > Math.abs(M[piv][c])) piv = r;
    [M[c], M[piv]] = [M[piv], M[c]];
    for (let r = c + 1; r < n; r++) {
      const f = M[r][c] / M[c][c];
      for (let k = c; k <= n; k++) M[r][k] -= f * M[c][k];
    }
  }
  const x = new Array(n).fill(0);
  for (let r = n - 1; r >= 0; r--) {
    let s = M[r][n];
    for (let k = r + 1; k < n; k++) s -= M[r][k] * x[k];
    x[r] = s / M[r][r];
  }
  return x;
}

    const row = body.insertRow();
    const th = document.createElement("th"); th.textContent = i + 1; row.appendChild(th);

      const td = row.insertCell();

      td.style.minWidth = cell + "px";
      const v = M[i][j];
      if (empty(i, j)) { td.className = "empty"; td.textContent = showNumbers ? "0" : ""; }
      else {
        const t = v / maxAbs;
        td.style.background = d3.interpolateRdBu(0.5 + 0.45 * t);
        td.style.color = Math.abs(t) > 0.55 ? "white" : "#111";
        if (showNumbers) td.textContent = fmt(v, 4);

      td.title = `K[${i + 1},${j + 1}] = ${fmt(v, 6)}`;

      td.onclick = () => onPick(i, j);
    }
    if (rowSums) {
      const td = row.insertCell(); td.className = "rowsum";
      td.style.minWidth = cell + "px";
      td.textContent = fmt(d3.sum(M[i]), 3);
    }
  }
  return table;
}

Part 1 — Lagrange polynomials

What they are

On the reference element \(\xi \in [-1, 1]\) we place \(p+1\) nodes \(\xi_1, \dots, \xi_{p+1}\). The Lagrange basis function attached to node \(i\) is

\[ N_i(\xi) \;=\; \prod_{\substack{k=1 \\ k\neq i}}^{p+1} \frac{\xi - \xi_k}{\xi_i - \xi_k}. \]

Two properties make them the workhorse of FEA:

  1. Kronecker delta: \(N_i(\xi_j) = \delta_{ij}\). Each numerator is zero at every other node, and the denominator is chosen so the value at node \(i\) is exactly 1. So the coefficient in front of \(N_i\) is literally the value of the field at node \(i\).
  2. Partition of unity: \(\sum_i N_i(\xi) = 1\) for every \(\xi\). So a constant field (a rigid-body translation) is reproduced exactly, and \(\sum_i N_i'(\xi) = 0\).

TipTry this
  • Highlight one \(N_i\): the big dot sits at height 1 on its own node and the others sit at 0. That’s the Kronecker delta property. Between the nodes, \(N_i\) can be negative or bigger than 1, and that’s fine.
  • Turn on Sum ΣNᵢ. However ugly the individual functions get, they always add to exactly 1.
  • Set \(p = 8\) with equally spaced nodes. The basis functions near the ends blow up (the Runge phenomenon). Switch to Gauss–Lobatto–Legendre and they calm down. This is why high-order (spectral) elements put their nodes at GLL points.
  • Turn on derivatives. At any \(\xi\) the derivatives add up to 0 (because \(\sum N_i = 1\)). That fact will show up again as the zero row sums of the stiffness matrix.

Interpolation: a field as a weighted sum of basis functions

In FEA we never store a function directly. We store nodal values \(u_i\) and rebuild the field as

\[ u_h(\xi) = \sum_{i=1}^{p+1} N_i(\xi)\, u_i . \]

TipTry this
  • In Manual mode, drag one \(u_i\). Only the \(N_i\)-shaped piece of the curve changes, and the curve still passes through every other node untouched.
  • Choose ξ³ − ξ and set \(p = 3\): the error drops to ~0. A degree-\(p\) Lagrange basis reproduces any polynomial of degree \(\le p\) exactly. That’s where FEA convergence rates come from.
  • Choose Runge with equally spaced nodes and raise \(p\) to 10–12. The error grows. Switch to GLL and it shrinks. In FEA we usually refine by adding more low-order elements (h-refinement), not by cranking up \(p\) on one element.

Part 2 — From weak form to stiffness matrix

The chain of ideas

Model problem: an axial bar of length \(L\) with stiffness \(EA(x)\) and distributed load \(f(x)\):

\[ -\frac{d}{dx}\!\left(EA\,\frac{du}{dx}\right) = f(x), \qquad u(0)=0, \qquad EA\,u'(L) = P . \]

Step 1: weak form. Multiply by a test function \(w\) (with \(w(0)=0\)), integrate over the bar, and integrate by parts:

\[ \int_0^L \frac{dw}{dx}\,EA\,\frac{du}{dx}\,dx \;=\; \int_0^L w\,f\,dx \;+\; w(L)\,P \qquad \text{for all admissible } w. \]

Step 2: Galerkin substitution. Replace both \(u\) and \(w\) with combinations of the same Lagrange basis functions:

\[ u_h(x) = \sum_{j} N_j(x)\,d_j, \qquad w_h(x) = \sum_{i} N_i(x)\,c_i . \]

Step 3: pull out the constants. \(d_j\) and \(c_i\) are just numbers, so they come out of the integral:

\[ \sum_i c_i \Bigg[ \sum_j \underbrace{\left(\int_0^L \frac{dN_i}{dx}\,EA\,\frac{dN_j}{dx}\,dx\right)}_{K_{ij}} d_j \;-\; \underbrace{\left(\int_0^L N_i f\,dx + N_i(L)P\right)}_{F_i} \Bigg] = 0 . \]

Because this must hold for every choice of \(c_i\), each bracket must vanish on its own, which gives \(\mathbf{K}\mathbf{d} = \mathbf{F}\).

NoteThe key picture
  • Row \(i\) of \(\mathbf K\) is the weak form “tested” with \(w = N_i\).
  • Column \(j\) of \(\mathbf K\) is how much the trial function \(N_j\) (the displacement shape of node \(j\)) contributes to that test.
  • \(K_{ij}\) is the area under the curve \(EA\,N_i'\,N_j'\). The interactive tools below let you look at that curve and its area directly.

Step 4: do it element by element. Each element is mapped to the reference element with \(x(\xi) = x_1^e + \tfrac{h}{2}(1+\xi)\), so \(J = \frac{dx}{d\xi} = \frac{h}{2}\) and, by the chain rule, \(\frac{dN}{dx} = \frac{1}{J}\frac{dN}{d\xi}\):

\[ K^e_{ab} = \int_{-1}^{1} EA(\xi)\,\frac{dN_a}{d\xi}\frac{dN_b}{d\xi}\,\frac{1}{J^2}\;J\,d\xi \;\approx\; \sum_{q=1}^{n_q} w_q\, EA(\xi_q)\,\frac{dN_a}{d\xi}(\xi_q)\,\frac{dN_b}{d\xi}(\xi_q)\,\frac{1}{J}. \]

TipTry this
  • p = 1, click (1,2). \(N_1'\) is negative and \(N_2'\) is positive, so the product is negative and \(K_{12} = -EA/h\). The off-diagonal terms of a stiffness matrix are negative because pulling one node drags its neighbour.
  • Click a diagonal cell. The integrand is \(EA\,(N_i')^2 \ge 0\), so diagonal entries are always positive.
  • Double h. The derivatives halve, their product drops by 4, and the interval doubles, so \(K \propto 1/h\) (a longer bar is softer).
  • Set p = 3 and drop to n_q = 1. The Gauss dot samples a single height, the entries go wrong, and the readout flags it. Turn on the EA slope α and watch the required \(n_q\) go up.
  • Check symmetry. \(K_{ij} = K_{ji}\) because \(N_i'N_j' = N_j'N_i'\). That is a direct result of using the same functions for test and trial (Galerkin).

Interactive 2 — Assembly: element matrices → global K

Each global basis function \(N_A(x)\) is stitched together from the local functions of the elements that share node \(A\). So a global entry \(K_{AB}\) is the sum of the element contributions over the elements where both \(N_A\) and \(N_B\) are nonzero.

Use the slider to add elements one at a time, then click any cell of the global matrix (zeros included).

Interactive 3 — Solve Kd = F and compare to the exact solution

Same mesh as above (\(EA = 1\), \(L = 1\)). The boundary condition \(u(0) = 0\) is applied by deleting row/column 1 (that removes the rigid-body mode that made \(\mathbf K\) singular). Then we solve.

viewof loadType = Inputs.radio(["Uniform  f = q", "Linear  f = q·x/L", "Sinusoidal  f = q·sin(πx/L)"], {value: "Sinusoidal  f = q·sin(πx/L)", label: "Distributed load"})
viewof qA = Inputs.range([-5, 5], {step: 0.5, value: 2, label: "q"})
viewof PA = Inputs.range([-3, 3], {step: 0.25, value: 0, label: "End load P at x = L"})
fLoad = loadType.startsWith("Uniform") ? (x => qA) : loadType.startsWith("Linear") ? (x => qA * x / LA) : (x => qA * Math.sin(Math.PI * x / LA))
uExact = {
  const L = LA, q = qA, P = PA, pi = Math.PI;
  if (loadType.startsWith("Uniform")) return {u: x => q * (L * x - x * x / 2) + P * x, du: x => q * (L - x) + P};
  if (loadType.startsWith("Linear")) return {u: x => -q * x ** 3 / (6 * L) + (P + q * L / 2) * x, du: x => -q * x * x / (2 * L) + P + q * L / 2};
  return {u: x => q * L * L / (pi * pi) * Math.sin(pi * x / L) + (P + q * L / pi) * x, du: x => q * L / pi * Math.cos(pi * x / L) + P + q * L / pi};
}
solution = {
  const K = d3.range(nDofA).map(() => new Array(nDofA).fill(0));

    for (let a = 0; a <= pA; a++) {

  const d = [0, ...solveDense(Kr, F.slice(1))];
  return {K, F, d};
}
